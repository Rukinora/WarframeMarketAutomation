/--------------------------------------------------------------------/
- Build Application with no Command Prompt:
- Paste the following code into the command prompt while inside of 
the Project Folder "WarframeMarketAutomation\"
/--------------------------------------------------------------------/



::Removes artifacts
cd src-tauri
cargo clean

cd ..

::Removes any old build files
rmdir /s /q "backend\build"
rmdir /s /q "backend\dist"
del /q /f "backend\main.spec"
del /q /f "src-tauri\bin\main.exe"

cd backend

::Installs all dependicies
pyinstaller --onefile --noconsole --collect-all selenium --collect-all webdriver_manager --collect-all keyboard main.py

move dist\main.exe ..\src-tauri\bin

cd ..

::builds app
npm run tauri build
